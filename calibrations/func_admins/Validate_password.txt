from django.core.exceptions import ValidationError
from django.contrib.auth.password_validation import validate_password
from django.core.validators import RegexValidator

class UserCreateSerializer(serializers.ModelSerializer):
    password = serializers.CharField(
        write_only=True,
        required=True,
        style={'input_type': 'password'},
        validators=[
            # Минимальная длина 8 символов
            RegexValidator(
                regex=r'^.{8,}$',
                message='Пароль должен содержать минимум 8 символов'
            ),
            # Хотя бы одна цифра
            RegexValidator(
                regex=r'\d',
                message='Пароль должен содержать хотя бы одну цифру'
            ),
            # Хотя бы одна буква
            RegexValidator(
                regex=r'[a-zA-Z]',
                message='Пароль должен содержать хотя бы одну букву'
            ),
            # Можно добавить другие правила (спецсимволы и т.д.)
        ]
    )
    group_id = serializers.IntegerField(required=True)

    class Meta:
        model = User
        fields = ('username', 'password', 'first_name', 'email', 'group_id')
        extra_kwargs = {
            'email': {'required': False, 'allow_blank': True}
        }

    # Добавляем валидацию пароля
    def validate_password(self, value):
        try:
            # Используем встроенную валидацию Django
            validate_password(value)
        except ValidationError as e:
            raise serializers.ValidationError(list(e.messages))
        return value

    # Остальные методы остаются без изменений
    def validate_group_id(self, value):
        try:
            with transaction.atomic():
                group = Group.objects.get(pk=value)
                if group.name not in ['admins', 'extusers', 'users']:
                    raise serializers.ValidationError("Недопустимая группа")
                return value
        except Group.DoesNotExist:
            raise serializers.ValidationError("Группа не существует")

    def validate(self, attrs):
        email = attrs.get('email')
        if email and User.objects.filter(email=email).exists():
            raise serializers.ValidationError({"email": "Этот email уже используется"})
        return attrs

    def create(self, validated_data):
        with transaction.atomic():        
            group_id = validated_data.pop('group_id')
            group = Group.objects.get(pk=group_id)
        
            request = self.context.get('request')
            if group.name == 'admins' and not request.user.is_superuser:
                raise serializers.ValidationError({"group_id": "Только superuser может создавать пользователей группы admins"})
            if not request.user.is_superuser and group.name not in ['extusers', 'users']:
                raise serializers.ValidationError({"group_id": "Вы можете создавать только пользователей групп extusers или users"})
            
            email = validated_data.get('email', '')
            if email == '':
                validated_data.pop('email', None)
        
            user = User.objects.create_user(**validated_data)
            user.groups.add(group)

            UserProfile.objects.update_or_create(
                user=user,
                defaults={
                    'is_verified': True,
                    'registration_date': timezone.now(),
                    'verification_date': timezone.now()
                }
            )

            ActionHistory.objects.create(
                user=request.user,
                action_type='user_create',
                description=f"Создан пользователь {user.username} с группой {group.name}",
                ip_address=request.META.get('REMOTE_ADDR')
            )
        
            return user

Использована встроенная валидация Django (validate_password):

Проверяет пароль на соответствие стандартным требованиям

Можно добавить дополнительные проверки через AUTH_PASSWORD_VALIDATORS в settings.py

Гибкая система валидации:

Можно легко добавлять/удалять правила через RegexValidator

Сообщения об ошибках четко указывают на проблему

Совместимость с существующим кодом:

Все остальные методы остаются без изменений

Валидация пароля выполняется до создания пользователя

Дополнительные рекомендации:
Для более сложных проверок можно создать кастомный валидатор:

python
def validate_password_complexity(value):
    if len(value) < 8:
        raise ValidationError("Пароль должен содержать минимум 8 символов")
    if not any(char.isdigit() for char in value):
        raise ValidationError("Пароль должен содержать хотя бы одну цифру")
    if not any(char.isalpha() for char in value):
        raise ValidationError("Пароль должен содержать хотя бы одну букву")
