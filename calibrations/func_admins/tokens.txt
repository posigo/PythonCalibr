Создаем новый permission в func_admins/permissions.py:

python
from rest_framework.permissions import BasePermission
from rest_framework_simplejwt.exceptions import TokenError
from rest_framework_simplejwt.tokens import RefreshToken

class IsTokenOwner(BasePermission):
    """
    Проверяет, что refresh token принадлежит текущему пользователю
    """
    message = 'Токен не принадлежит текущему пользователю'

    def has_permission(self, request, view):
        if request.method == 'POST' and view.action == 'logout':
            refresh_token = request.data.get('refresh')
            
            if not refresh_token:
                return False

            try:
                token = RefreshToken(refresh_token)
                return token.payload.get('user_id') == request.user.id
            except TokenError:
                return False
        
        return True

from rest_framework_simplejwt.exceptions import InvalidToken
from rest_framework_simplejwt.tokens import RefreshToken
from rest_framework.response import Response
from rest_framework import status

class TokenOwnershipMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        # Проверяем только запросы к эндпоинтам с токенами
        if request.path in ['/api/auth/logout/', '/api/token/refresh/']:
            if request.method == 'POST' and 'refresh' in request.data:
                try:
                    token = RefreshToken(request.data['refresh'])
                    if token.payload.get('user_id') != request.user.id:
                        return Response(
                            {'detail': 'Токен не принадлежит пользователю'},
                            status=status.HTTP_403_FORBIDDEN
                        )
                except InvalidToken:
                    return Response(
                        {'detail': 'Недействительный токен'},
                        status=status.HTTP_400_BAD_REQUEST
                    )

        return self.get_response(request)

Лучшее решение - вынести проверку в отдельный метод сервисного слоя, а затем использовать его и в permission классе, и в view. Вот как это правильно реализовать:

Создаем сервис для работы с токенами (func_admins/services/token_service.py):

python
from rest_framework_simplejwt.exceptions import TokenError
from rest_framework_simplejwt.tokens import RefreshToken
from rest_framework import status
from django.contrib.auth import get_user_model

User = get_user_model()

class TokenService:
    @staticmethod
    def validate_refresh_token(refresh_token: str, user: User) -> dict:
        """
        Валидация refresh token и проверка принадлежности пользователю
        Возвращает словарь с результатами проверки:
        {
            'is_valid': bool,
            'token': RefreshToken | None,
            'error_message': str | None,
            'status_code': int
        }
        """
        # 1. Проверка наличия токена
        if not refresh_token:
            return {
                'is_valid': False,
                'error_message': 'Refresh token обязателен',
                'status_code': status.HTTP_400_BAD_REQUEST
            }

        # 2. Проверка формата токена
        if not isinstance(refresh_token, str) or not refresh_token.strip():
            return {
                'is_valid': False,
                'error_message': 'Неверный формат refresh token',
                'status_code': status.HTTP_400_BAD_REQUEST
            }

        try:
            # 3. Валидация токена
            token = RefreshToken(refresh_token)
            
            # 4. Проверка принадлежности токена пользователю
            if token.payload.get('user_id') != user.id:
                return {
                    'is_valid': False,
                    'error_message': 'Токен не принадлежит текущему пользователю',
                    'status_code': status.HTTP_403_FORBIDDEN
                }

            return {
                'is_valid': True,
                'token': token,
                'error_message': None,
                'status_code': None
            }

        except TokenError as e:
            return {
                'is_valid': False,
                'error_message': f'Недействительный refresh token: {str(e)}',
                'status_code': status.HTTP_400_BAD_REQUEST
            }

from .services.token_service import TokenService

class IsValidRefreshTokenOwner(BasePermission):
    """
    Проверяет что refresh token валиден и принадлежит текущему пользователю
    """
    message = 'Недействительный токен или токен другого пользователя'

    def has_permission(self, request, view):
        if request.method == 'POST' and view.action == 'logout':
            refresh_token = request.data.get('refresh')
            validation_result = TokenService.validate_refresh_token(refresh_token, request.user)
            
            if not validation_result['is_valid']:
                self.message = validation_result['error_message']
            
            return validation_result['is_valid']
        return True

from .services.token_service import TokenService
from .permissions import IsValidRefreshTokenOwner

class AuthViewSet(viewsets.GenericViewSet):
    @action(
        detail=False,
        methods=['post'],
        permission_classes=[IsAuthenticated, IsValidRefreshTokenOwner]  # Добавляем permission
    )
    def logout(self, request):
        try:
            with transaction.atomic():
                refresh_token = request.data.get('refresh')
                
                # Повторная проверка через сервис (для единообразия)
                validation_result = TokenService.validate_refresh_token(refresh_token, request.user)

                    {'detail': 'Успешный выход из системы.'},

from django.core.cache import cache

class TokenService:
    @staticmethod
    def is_token_blacklisted(token):
        cache_key = f"blacklisted_token_{token}"
        if cache.get(cache_key):
            return True
            
        is_blacklisted = BlacklistedToken.objects.filter(token__token=token).exists()
        cache.set(cache_key, is_blacklisted, timeout=300)  # Кэш на 5 минут
        return is_blacklisted
